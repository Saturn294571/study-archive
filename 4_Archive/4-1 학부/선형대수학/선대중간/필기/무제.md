$$\begin{equation*}
\begin{pmatrix}
2 & 7 & 1 \\
8 & 2 & 8 \\
1 & 8 & 2
\end{pmatrix} = LU
\end{equation*}$$

$$
U: \begin{pmatrix}
2 & 7 & 1 \\
8 & 2 & 8 \\
1 & 8 & 2
\end{pmatrix} \rightarrow
\begin{pmatrix}
2 & 7 & 1 \\
0 & 26 & 0 \\
0 & 9 & 3
\end{pmatrix} \rightarrow
\begin{pmatrix}
2 & 7 & 1 \\
0 & 26 & 0 \\
0 & 0 & \frac{26}{3}
\end{pmatrix} = U
$$

$$
L: \begin{pmatrix}
2 & 0 & 0 \\
8 & 26 & 0 \\
1 & 9 & 26/3
\end{pmatrix} \rightarrow
\begin{pmatrix}
1 & 0 & 0 \\
4 & 1 & 0 \\
\frac{1}{2} & \frac{9}{26} & 1
\end{pmatrix} = L
$$
